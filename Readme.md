
# Install build dependencies    
(for Fedora)    
```
sudo dnf install SDL3-devel \
    gcc \
    gcc-c++ \
    cmake \
    make \
    pkgconf-pkg-config \
    mesa-libGL-devel
```

further window-system development dependencies:    
```
sudo dnf install \
    libX11-devel \
    libXext-devel \
    libXrandr-devel \
    libXcursor-devel \
    libXfixes-devel \
    libXi-devel \
    libXScrnSaver-devel \
    libXtst-devel \
    libxkbcommon-devel \
    wayland-devel \
    wayland-protocols-devel \
    libdecor-devel \
    mesa-libGL-devel
```
